INTERFACE PRACTICE 1



Static Methods

Introduced in Java 8 (2014).
These are methods defined in an interface using the static keyword. They belong to the interface itself and can be called using the interface name.



Default Methods

Introduced in Java 8 (2014).
These are instance methods defined in an interface using the default keyword. 
They provide a default implementation, which classes implementing the interface can override.






Private Methods

Introduced in Java 9 (2017).
These methods are used within the interface to encapsulate common functionality shared by default or static methods.

import java.util.*;

interface InterfaceOne {
    void add();// implictly will create public modifier
    void read();
    int work();
    
    default void playing()
    {
        System.out.println("Playing");
    }
}

class Person implements InterfaceOne {
    public void add() {
        System.out.println("Person is adding money");
    }

    public void read() {
        System.out.println("Person is reading money");
    }

    public int work() {
        System.out.println("Person is working in office");
        return 0;
    }
    
    // public void playing()
    // {
    //     System.out.println("Not Playing"); 
    // }
}

public class Main { 
    public static void main(String[] args) {
        Person person = new Person();
        person.add();
        person.read();
        person.work();
        person.playing();//---if not overrides will call interface default method otherwise calls class method 
        
    }
}
